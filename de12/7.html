ゲーム作成

MANTIS　ATTACK

真ん中にいるカマキリが両側から歩いてくる棒人間を倒すゲーム。AとDボタンで左右の鎌を振り落としRボタンでゲームをリスタートすることができる。

ゲームの始め方

このソースコードをコピペしターミナルに「python -m pip install pygame」→「python mantis_attack.py」の順に貼り付けて開始する。

279def reset_game():
280    global enemies, score, game_over
281    global spawn_timer, elapsed, attacks
282
283    enemies = []
284    score = 0
285    game_over = False
286    spawn_timer = 0
287    elapsed = 0
288
289    attacks = {
290        "left": 0,
291        "right": 0
292    }

298
299# --------------------
300# メインループ
301# --------------------
302running = True
303
304while running:
305
306    dt = clock.tick(60) / 1000
307
308    # 背景
309    screen.fill(WHITE)
310
311    # 地面
312    pygame.draw.line(
313        screen, GRAY,
314        (0, GROUND + 48),
315        (WIDTH, GROUND + 48), 3
316    )
317
318    # --------------------
319    # イベント処理
320    # --------------------
321    for event in pygame.event.get():
322
323        if event.type == pygame.QUIT:
324            running = False
325
326        # キーボード
327        if event.type == pygame.KEYDOWN:
328
329            if not started:
330                if event.key == pygame.K_RETURN:
331                    reset_game()
332                    started = True
333
334            elif not game_over:
335
336                if event.key in (pygame.K_LEFT, pygame.K_a):
337                    attack_enemy("left")
338
339                if event.key in (pygame.K_RIGHT, pygame.K_d):
340                    attack_enemy("right")
341
342            elif event.key == pygame.K_r:
343                reset_game()
344
345        # マウス
346        if event.type == pygame.MOUSEBUTTONDOWN:
347
348            if event.button == 1:
349
350                if not started:
351
352                    if start_button.collidepoint(event.pos):
353                        reset_game()
354                        started = True
355
356                elif game_over:
357                    reset_game()
358
359                else:
360
361                    if left_button.collidepoint(event.pos):
362                        attack_enemy("left")
363
364                    elif right_button.collidepoint(event.pos):
365                        attack_enemy("right")
366
367    # --------------------
368    # ゲーム進行
369    # --------------------
370    if started and not game_over:
371
372        elapsed += dt
373        spawn_timer += dt
374
375        # 敵の出現
376        spawn_interval = max(0.55, 1.5 - elapsed * 0.015)
377
378        if spawn_timer >= spawn_interval:
379            spawn_enemy()
380            spawn_timer = 0
381
382        # 攻撃時間
383        for side in attacks:
384            if attacks[side] > 0:
385                attacks[side] = max(0, attacks[side] - dt)
386
387        # 敵の移動と攻撃判定
388        for enemy in enemies[:]:
389
390            if enemy["side"] == "left":
391                enemy["x"] += enemy["speed"] * dt
392            else:
393                enemy["x"] -= enemy["speed"] * dt
394
395            side = enemy["side"]
396
397            if (
398                attacks[side] > 0
399                and attacks[side] <= ATTACK_HIT_TIME
400                and abs(enemy["x"] - CENTER) < 205
401            ):
402                enemies.remove(enemy)
403                score += 1
404                continue
405
406            # 中央に到達したらゲームオーバー
407            if abs(enemy["x"] - CENTER) < 65:
408                game_over = True
409
410    # --------------------
411    # 画面描画
412    # --------------------
413
414    if not started:
415
416        # タイトル画面
417        draw_title()
418
419    else:
420
421        # 敵
422        for enemy in enemies:
423            draw_enemy(enemy)
424
425        # カマキリ
426        draw_mantis()
427
428        # スコア
429        score_text = font.render(
430            f"SCORE: {score}", True, BLACK
431        )
432        screen.blit(score_text, (30, 25))
433
434        # 時間
435        time_text = font.render(
436            f"TIME: {int(elapsed)}", True, BLACK
437        )
438        screen.blit(time_text, (WIDTH - 180, 25))
439
440        # 攻撃ボタン
441        draw_button(left_button, BLUE, "LEFT ATTACK")
442        draw_button(right_button, RED, "RIGHT ATTACK")
443
444        # ゲームオーバー
445        if game_over:
446
447            overlay = pygame.Surface((WIDTH, HEIGHT))
448            overlay.set_alpha(170)
449            overlay.fill(WHITE)
450            screen.blit(overlay, (0, 0))
451
452            title = big_font.render(
453                "GAME OVER", True, RED
454            )
455
456            screen.blit(
457                title,
458                title.get_rect(center=(CENTER, 230))
459            )
460
461            result = font.render(
462                f"FINAL SCORE: {score}", True, BLACK
463            )
464
465            screen.blit(
466                result,
467                result.get_rect(center=(CENTER, 310))
468            )
469
470            restart = font.render(
471                "TAP TO RESTART / PRESS R", True, BLACK
472            )
473
474            screen.blit(
475                restart,
476                restart.get_rect(center=(CENTER, 390))
477            )
478
479    # 画面更新
480    pygame.display.flip()
